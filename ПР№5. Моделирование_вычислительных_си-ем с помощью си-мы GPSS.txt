Программа GPSS для Задачи 5.1 

010  MEM        STORAGE    8
020  ZEXP       FUNCTION   RN1,C12
                0,0/.2,.22/.4,.51/.5,.69/.6,.92/.7,1.2/.8,1.61
                .9,2.3/.95,3/.99,4.6/.999,6.9/1,100
030  GENERATE   (Exponential(1,0,2))   ;  
040  QUEUE      A11
041  TEST GE    Q$A11,5,EXIT           ;    
050  ENTER      MEM,1
060  DEPART     A11
070  TABULATE   Q1                     ;    
080  MET6       QUEUE    A2
090  TRANSFER   BOTH,MET1,MET2
100  MET1       SEIZE    CPU1
110  DEPART     A2
120  ADVANCE    5,FNSZEXP              ;   
130  RELEASE    CPU1
140  TRANSFER   ,MET3
150  MET2       SEIZE    CPU2
160  DEPART     A2
170  ADVANCE    2,FNSZEXP              ; 
180  RELEASE    CPU2
190  MET3       TRANSFER .6,MET5,MET4  ; 
200  MET4       QUEUE    A3
210  SEIZE      DISK
220  DEPART     A3
230  ADVANCE    5,3                    ;    
240  RELEASE    DISK
250  TRANSFER   ,MET6                  ; 
260  MET5       LEAVE    MEM,1
270  TERMINATE  1

300  START      500                    ;
